(version 1)
; Every denial carries this sandbox's tag, which is how the denials of one sandbox
; are told apart from every other sandboxed process's in the system log.
(deny default {{ audit }})

; =============================================================================
; RULE ORDER MATTERS
;
; SBPL resolves an operation with the LAST matching rule, not the first. This
; profile is therefore laid out in three passes:
;
;   1. baseline    - broad allows that make a process able to run at all
;   2. protections - denies for user data and for writable-then-executable
;                    locations
;   3. user rules  - explicitly configured paths, which come last so that an
;                    explicit allow beats the protections of pass 2
;   4. host        - credential stores and keys, and persistence locations. They
;                    follow every grant, so a grant of a directory that merely
;                    contains one cannot reopen it. For a credential location
;                    only, a grant naming it, or a path inside it, does.
;
; Moving a rule between passes changes what the sandbox enforces.
; =============================================================================

; =============================================================================
; PASS 1: BASELINE
; =============================================================================

; Baseline exec. Pass 2 removes exec from every location the sandboxed process
; can write, which is what stops it from dropping a payload and running it.
; Reads are needed to exec, and macOS's runtime layout (dyld caches, cryptexes,
; sealed system volumes) is not stably enumerable across releases, so both start
; broad and are narrowed by the denies below.
(allow process-exec)
(allow file-read*)

{% if writable_file_system %}
; Permissive mode: the whole filesystem is writable.
(allow file-write*)
{% endif %}

{% if !filesystem_strict %}
; --- Shared temp directories ---
; Readable outside strict mode, but writable only when the whole filesystem is.
; The sandbox has its own writable directory, exported as TMPDIR, so nothing
; needs to write here.
(allow file-read* (subpath "/private/tmp"))
(allow file-read* (subpath "/private/var/folders"))
{% endif %}

(allow file-read* (literal "/dev/tty"))
(allow file-read* (regex #"^/dev/ttys[0-9]+$"))
(allow file-read* (regex #"^/dev/pty[pq][0-9a-f]+$"))
(allow file-write* (literal "/dev/null"))
(allow file-write* (literal "/dev/zero"))

{% if allow_tty_write %}

; =============================================================================
; PASS 2: PROTECTIONS
;
; Patterns are anchored to directory or file boundaries so that "/\.ssh(/|$)"
; matches "/.ssh/" and a trailing "/.ssh" but not "not.ssh.txt". The protections
; of the credential toggles are in pass 4.
; =============================================================================

; --- No exec from anywhere the sandboxed process can write ---
; Without these, the baseline exec allow above would let sandboxed code write a
; binary to a scratch directory and run it. Permissive mode makes the whole
; filesystem writable and so gives up this guarantee by definition; the scratch
; directories stay denied there anyway, since that costs nothing.
(deny process-exec (subpath "/private/tmp"))
(deny process-exec (subpath "/private/var/folders"))

{% if filesystem_strict %}
; --- Strict mode: no reads of user data ---
; System files stay readable because the macOS runtime layout is not stably
; enumerable, but everything owned by a user is denied unless explicitly
; allowed in pass 3.
(deny file-read* (subpath "/Users") {{ audit }})
(deny file-read* (subpath "/home") {{ audit }})
(deny file-read* (subpath "/Volumes") {{ audit }})
(deny file-read* (subpath "/Network") {{ audit }})
(deny file-read* (subpath "/private/var/root") {{ audit }})
(deny file-read* (subpath "/private/tmp") {{ audit }})
(deny file-read* (subpath "/private/var/folders") {{ audit }})
{% endif %}
{% if protect_user_home %}
{% if allow_tcc_prompts %}
; TCC prompts enabled: only ~/Library is blocked here. The TCC-protected
; folders (Desktop, Documents, Downloads, Pictures, Movies, Music) are left to
; macOS, which prompts the user for access.
(deny file-read* (regex #"/Users/[^/]+/Library(/|$)") {{ audit }})
{% else %}
; TCC prompts disabled: block the sensitive home subdirectories outright.
(deny file-read* (regex #"/Users/[^/]+/Documents(/|$)") {{ audit }})
(deny file-read* (regex #"/Users/[^/]+/Desktop(/|$)") {{ audit }})
(deny file-read* (regex #"/Users/[^/]+/Downloads(/|$)") {{ audit }})
(deny file-read* (regex #"/Users/[^/]+/Library(/|$)") {{ audit }})
(deny file-read* (regex #"/Users/[^/]+/Pictures(/|$)") {{ audit }})
(deny file-read* (regex #"/Users/[^/]+/Movies(/|$)") {{ audit }})
(deny file-read* (regex #"/Users/[^/]+/Music(/|$)") {{ audit }})
{% endif %}

; =============================================================================
; PASS 3: USER-CONFIGURED PATHS
;
; These come last so that an explicitly configured path wins over the
; protections above. The credential and persistence protections of pass 4 still
; follow them. Every path here is absolute, already resolved and verified by the
; profile generator to contain no link: SBPL matches resolved paths, so
; "/tmp/x" would never match the kernel's "/private/tmp/x".
; =============================================================================

; Ancestors of every granted path, so that resolving into an allowed directory
; is not blocked by a protection rule covering one of its parents. These grant
; the directory itself, not its other children.
{% for path in traversal_paths %}
(allow file-read* (literal "{{ path }}"))
{% endfor %}

; Configured grants, one block each. Read is always granted; write is granted
; when the grant allows it; and exec is settled last, so within a block the
; last matching rule is the one the grant asks for. A writable path that is not
; also executable is denied exec, which is what stops sandboxed code from
; staging a payload; a path granted both is executable on purpose, and gives up
; that guarantee for itself alone.
;
; A grant naming a directory is rendered with `subpath` and covers its whole
; tree; one naming a file is rendered with `literal`. The profile generator
; decides which from the path itself.
{% for grant in grants %}
(allow file-read* ({{ grant.filter }} "{{ grant.path }}"))
{% if grant.writable %}
(allow file-write* ({{ grant.filter }} "{{ grant.path }}"))
(allow file-write-create ({{ grant.filter }} "{{ grant.path }}"))
(allow file-write-unlink ({{ grant.filter }} "{{ grant.path }}"))
{% endif %}
{% if grant.executable %}
(allow process-exec ({{ grant.filter }} "{{ grant.path }}"))
{% else if grant.writable %}
(deny process-exec ({{ grant.filter }} "{{ grant.path }}"))
{% endif %}
{% endfor %}

; Working directory: full read and write, no exec. It comes after the grants
; above on purpose — the sandbox's own scratch directory is never executable,
; not even when a grant covering one of its parents asks for exec.
(allow file-read* (subpath "{{ working_dir }}"))
(allow file-write* (subpath "{{ working_dir }}"))
(allow file-write-create (subpath "{{ working_dir }}"))
(allow file-write-unlink (subpath "{{ working_dir }}"))
(deny process-exec (subpath "{{ working_dir }}"))

{% if let Some(bin_dir) = ipc_bin_dir %}
; IPC wrapper scripts are generated by the host before the sandbox starts and
; are the one exception to the no-exec-in-the-working-directory rule.
(allow process-exec (subpath "{{ bin_dir }}"))
{% endif %}
{% if let Some(socket_dir) = ipc_socket_dir %}
; The IPC socket lives outside the working directory, so that a sandboxed
; process cannot unlink the endpoint it depends on. Connecting to a Unix socket
; needs read and write on the socket file itself.
(allow file-read* (subpath "{{ socket_dir }}"))
(allow file-write* (subpath "{{ socket_dir }}"))
{% endif %}
{% if let Some(socket) = ipc_socket %}
; Connecting to a Unix socket is a network operation in SBPL, so file rules
; alone are not enough to reach the IPC endpoint.
(allow network-outbound (literal "{{ socket }}"))
{% endif %}

{% if let Some(venv) = python_venv %}
; Python virtual environment
(allow file-read* (subpath "{{ venv.path }}"))
(allow process-exec (subpath "{{ venv.path }}"))
{% if venv.writable %}
(allow file-write* (subpath "{{ venv.path }}"))
(allow file-write-create (subpath "{{ venv.path }}"))
(allow file-write-unlink (subpath "{{ venv.path }}"))
{% endif %}
{% endif %}

; =============================================================================
; PASS 4: HOST PROTECTION
;
; These come after every grant, so a grant of a directory that contains one
; cannot reopen it. A granted path that is, or lies inside, a credential location
; is the exception: it is left out of the rule that protects the location. The
; denials are the tagged kind so that they are reported.
; =============================================================================

; Sensitive locations the security toggles protect, wherever they are found.
{% for filter in protected %}
(deny file-read* {{ filter }} {{ audit }})
{% endfor %}
{% if let Some(read) = read_rules %}
; Credential stores and keys: not readable, and not movable out of the way, since
; a directory renamed to a new name would be readable again under it.
{% for filter in read.tree_filters %}
(deny file-read* {{ filter }} {{ audit }})
{% endfor %}
{% for tree in read.trees %}
(deny file-write-unlink (subpath "{{ tree }}") {{ audit }})
{% endfor %}
{% for parent in read.guarded_parents %}
(deny file-write-unlink (literal "{{ parent }}") {{ audit }})
{% endfor %}
; Agent sockets: not readable and not connectable, whatever the network policy.
(deny file-read* (regex #"^/private/tmp/com\.apple\.launchd\.[^/]+/Listeners$") {{ audit }})
(deny network-outbound (remote unix-socket (path-regex #"^/private/tmp/com\.apple\.launchd\.[^/]+/Listeners$")) {{ audit }})
{% for socket in read.sockets %}
(deny network-outbound (remote unix-socket (path-literal "{{ socket }}")) {{ audit }})
{% endfor %}
{% endif %}
{% if let Some(write) = write_rules %}
; Persistence locations: not writable, and the directories above them not
; removable, since a prepared lookalike moved into their name would stand in.
{% for tree in write.trees %}
(deny file-write* (subpath "{{ tree }}") {{ audit }})
{% endfor %}
{% for filter in write.anywhere %}
(deny file-write* {{ filter }} {{ audit }})
{% endfor %}
{% for parent in write.guarded_parents %}
(deny file-write-unlink (literal "{{ parent }}") {{ audit }})
{% endfor %}
{% for filter in write.anywhere_parents %}
(deny file-write-unlink {{ filter }} {{ audit }})
{% endfor %}
{% endif %}

; =============================================================================
; Network
;
; The IPC endpoint is a Unix socket, and connecting to one is a network
; operation in SBPL, so it is granted in pass 3 and not here. That grant
; survives the blanket deny below, which only governs IP sockets: removing it
; breaks IPC with EPERM even though it looks redundant next to this rule.
; =============================================================================
(deny network*)
; Sandboxed tools may host ephemeral loopback services for local previews,
; language servers and similar terminal-first workflows. macOS requires
; network-inbound here; that operation also grants the bind.
(allow network-inbound (local tcp "localhost:*"))
{% if let Some(port) = proxy_port %}
; Outbound traffic is restricted to the sandbox proxy, which applies the
; configured network policy to every request.
(allow network-outbound (remote ip "localhost:{{ port }}"))
{% endif %}
