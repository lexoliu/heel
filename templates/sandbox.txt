(version 1)
(deny default)

; =============================================================================
; RULE ORDER MATTERS
;
; SBPL resolves an operation with the LAST matching rule, not the first. This
; profile is therefore laid out in three passes:
;
;   1. baseline    - broad allows that make a process able to run at all
;   2. protections - denies for sensitive data and for writable-then-executable
;                    locations
;   3. user rules  - explicitly configured paths, which come last so that an
;                    explicit allow always beats a default protection
;
; Moving a rule between passes changes what the sandbox enforces.
; =============================================================================

; =============================================================================
; PASS 1: BASELINE
; =============================================================================

; Baseline exec. Pass 2 removes exec from every location the sandboxed process
; can write, which is what stops it from dropping a payload and running it.
; Reads are needed to exec, and macOS's runtime layout (dyld caches, cryptexes,
; sealed system volumes) is not stably enumerable across releases, so both start
; broad and are narrowed by the denies below.
(allow process-exec)
(allow file-read*)

{% if writable_file_system %}
; Permissive mode: the whole filesystem is writable.
(allow file-write*)
{% endif %}

{% if !filesystem_strict %}
; --- Shared temp directories ---
; Readable outside strict mode, but writable only when the whole filesystem is.
; The sandbox has its own writable directory, exported as TMPDIR, so nothing
; needs to write here.
(allow file-read* (subpath "/private/tmp"))
(allow file-read* (subpath "/private/var/folders"))
{% endif %}

(allow file-read* (subpath "/dev/fd"))
(allow file-read* (literal "/dev/tty"))
(allow file-read* (regex #"^/dev/ttys[0-9]+$"))
(allow file-read* (regex #"^/dev/pty[pq][0-9a-f]+$"))
(allow file-write* (literal "/dev/null"))
(allow file-write* (literal "/dev/zero"))
(allow file-write* (subpath "/dev/fd"))
{% if allow_tty_write %}

; =============================================================================
; PASS 2: PROTECTIONS
;
; Patterns are anchored to directory or file boundaries so that "/\.ssh(/|$)"
; matches "/.ssh/" and a trailing "/.ssh" but not "not.ssh.txt".
; =============================================================================

; --- No exec from anywhere the sandboxed process can write ---
; Without these, the baseline exec allow above would let sandboxed code write a
; binary to a scratch directory and run it. Permissive mode makes the whole
; filesystem writable and so gives up this guarantee by definition; the scratch
; directories stay denied there anyway, since that costs nothing.
(deny process-exec (subpath "/private/tmp"))
(deny process-exec (subpath "/private/var/folders"))

{% if filesystem_strict %}
; --- Strict mode: no reads of user data ---
; System files stay readable because the macOS runtime layout is not stably
; enumerable, but everything owned by a user is denied unless explicitly
; allowed in pass 3.
(deny file-read* (subpath "/Users") (with no-log))
(deny file-read* (subpath "/home") (with no-log))
(deny file-read* (subpath "/Volumes") (with no-log))
(deny file-read* (subpath "/Network") (with no-log))
(deny file-read* (subpath "/private/var/root") (with no-log))
(deny file-read* (subpath "/private/tmp") (with no-log))
(deny file-read* (subpath "/private/var/folders") (with no-log))
{% endif %}

; =============================================================================
; PASS 3: USER-CONFIGURED PATHS
;
; These come last so that an explicitly configured path always wins over the
; protections above. Every path here has been canonicalized by the profile
; generator: SBPL matches resolved paths, so "/tmp/x" would never match the
; kernel's "/private/tmp/x".
; =============================================================================

; Ancestors of every granted path, so that resolving into an allowed directory
; is not blocked by a protection rule covering one of its parents. These grant
; the directory itself, not its other children.
{% for path in traversal_paths %}
(allow file-read* (literal "{{ path }}"))
{% endfor %}

; Readable paths
{% for path in readable_paths %}
(allow file-read* (subpath "{{ path }}"))
{% endfor %}

; Writable paths, with exec removed so they cannot be used to stage a payload
{% for path in writable_paths %}
(allow file-read* (subpath "{{ path }}"))
(allow file-write* (subpath "{{ path }}"))
(allow file-write-create (subpath "{{ path }}"))
(allow file-write-unlink (subpath "{{ path }}"))
(deny process-exec (subpath "{{ path }}"))
{% endfor %}

; Working directory: full read and write, no exec
(allow file-read* (subpath "{{ working_dir }}"))
(allow file-write* (subpath "{{ working_dir }}"))
(allow file-write-create (subpath "{{ working_dir }}"))
(allow file-write-unlink (subpath "{{ working_dir }}"))
(deny process-exec (subpath "{{ working_dir }}"))

{% if let Some(bin_dir) = ipc_bin_dir %}
; IPC wrapper scripts are generated by the host before the sandbox starts and
; are the one exception to the no-exec-in-the-working-directory rule.
(allow process-exec (subpath "{{ bin_dir }}"))
{% endif %}
{% if let Some(socket_dir) = ipc_socket_dir %}
; The IPC socket lives outside the working directory, so that a sandboxed
; process cannot unlink the endpoint it depends on. Connecting to a Unix socket
; needs read and write on the socket file itself.
(allow file-read* (subpath "{{ socket_dir }}"))
(allow file-write* (subpath "{{ socket_dir }}"))
{% endif %}
{% if let Some(socket) = ipc_socket %}
; Connecting to a Unix socket is a network operation in SBPL, so file rules
; alone are not enough to reach the IPC endpoint.
(allow network-outbound (literal "{{ socket }}"))
{% endif %}

{% if let Some(venv) = python_venv %}
; Python virtual environment
(allow file-read* (subpath "{{ venv.path }}"))
(allow process-exec (subpath "{{ venv.path }}"))
{% if venv.writable %}
(allow file-write* (subpath "{{ venv.path }}"))
(allow file-write-create (subpath "{{ venv.path }}"))
(allow file-write-unlink (subpath "{{ venv.path }}"))
{% endif %}
{% endif %}

; Explicitly permitted executables. A grant naming a directory is rendered with
; the `subpath` filter and so covers the whole tree; a grant naming a file keeps
; `literal`. Both come after the writable denies above, so a directory that is
; both writable and executable ends up executable — that is the point of asking
; for it, and it gives up the write-then-execute guarantee for that tree.
{% for rule in executable_paths %}
(allow file-read* ({{ rule.filter }} "{{ rule.path }}"))
(allow process-exec ({{ rule.filter }} "{{ rule.path }}"))
{% endfor %}

; =============================================================================
; Network
;
; The IPC endpoint is a Unix socket, and connecting to one is a network
; operation in SBPL, so it is granted in pass 3 and not here. That grant
; survives the blanket deny below, which only governs IP sockets: removing it
; breaks IPC with EPERM even though it looks redundant next to this rule.
; =============================================================================
(deny network*)
; Sandboxed tools may host ephemeral loopback services for local previews,
; language servers and similar terminal-first workflows. macOS requires
; network-inbound here; that operation also grants the bind.
(allow network-inbound (local tcp "localhost:*"))
{% if let Some(port) = proxy_port %}
; Outbound traffic is restricted to the sandbox proxy, which applies the
; configured network policy to every request.
(allow network-outbound (remote ip "localhost:{{ port }}"))
{% endif %}
